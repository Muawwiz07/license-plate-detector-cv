    "def detect_license_plate(original, edges):\n",
    "    \"\"\"Find license plate using contour detection.\"\"\"\n",
    "    contours, _ = cv2.findContours(\n",
    "        edges.copy(), cv2.RETR_LIST, cv2.CHAIN_APPROX_SIMPLE\n",
    "    )\n",
    "    # Sort by area (largest first)\n",
    "    contours = sorted(contours, key=cv2.contourArea, reverse=True)[:30]\n",
    "\n",
    "    for contour in contours:\n",
    "        perimeter = cv2.arcLength(contour, closed=True)\n",
    "        approx = cv2.approxPolyDP(\n",
    "            contour, epsilon=0.018 * perimeter, closed=True\n",
    "        )\n",
    "\n",
    "        # Must be a rectangle (4 corners)\n",
    "        if len(approx) != 4:\n",
    "            continue\n",
    "\n",
    "        x, y, w, h = cv2.boundingRect(approx)\n",
    "        area = w * h\n",
    "        aspect_ratio = w / float(h)\n",
    "\n",
    "        # Aspect ratio filter: plates are wider than tall\n",
    "        if not (2.0 <= aspect_ratio <= 6.0):\n",
    "            continue\n",
    "\n",
    "        # Area filter: reasonable size\n",
    "        image_area = original.shape[0] * original.shape[1]\n",
    "        if not (0.005 * image_area <= area <= 0.25 * image_area):\n",
    "            continue\n",
    "\n",
    "        # Solidity filter: must be a solid rectangle\n",
    "        solidity = cv2.contourArea(contour) / float(area)\n",
    "        if solidity < 0.4:\n",
    "            continue\n",
    "\n",
    "        # Crop with padding\n",
    "        pad = 5\n",
    "        x1 = max(0, x - pad)\n",
    "        y1 = max(0, y - pad)\n",
    "        x2 = min(original.shape[1], x + w + pad)\n",
    "        y2 = min(original.shape[0], y + h + pad)\n",
    "\n",
    "        return original[y1:y2, x1:x2], (x, y, w, h), approx\n",
    "\n",
    "    return None, None, None\n",
    "\n",
    "\n",
    "# Run detection\n",
    "plate_img, bbox, contour = detect_license_plate(image, edges)\n",
    "\n",
    "if plate_img is not None:\n",
    "    x, y, w, h = bbox\n",
    "    print(f'✅ Plate detected!')\n",
    "    print(f'   Location : x={x}, y={y}')\n",
    "    print(f'   Size     : {w}x{h} pixels')\n",
    "    print(f'   Aspect   : {w/h:.2f}')\n",
    "\n",
    "    # Show detection result\n",
    "    debug_img = image.copy()\n",
    "    cv2.drawContours(debug_img, [contour], -1, (255, 0, 0), 2)\n",
    "    cv2.rectangle(debug_img, (x, y), (x+w, y+h), (0, 255, 0), 3)\n",
    "\n",
    "    fig, axes = plt.subplots(1, 2, figsize=(14, 5))\n",
    "    fig.suptitle('🔍 Plate Detection', fontsize=16, fontweight='bold')\n",

    "\n",

    "def extract_text(plate_img):\n",
    "    \"\"\"Run Tesseract OCR on plate crop.\"\"\"\n",
    "    # Grayscale\n",
    "    if len(plate_img.shape) == 3:\n",
    "        gray = cv2.cvtColor(plate_img, cv2.COLOR_BGR2GRAY)\n",
    "    else:\n",
    "        gray = plate_img.copy()\n",
    "\n",
    "    # Upscale for better OCR\n",
    "    scale = max(1, 100 // gray.shape[0] + 1)\n",
    "    if scale > 1:\n",
    "        gray = cv2.resize(gray, None, fx=scale, fy=scale,\n",
    "                          interpolation=cv2.INTER_CUBIC)\n",
    "\n",
    "    # Otsu threshold\n",
    "    _, binary = cv2.threshold(\n",
    "        gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU\n",
    "    )\n",
    "\n",
    "    config = (\n",
    "        r'--oem 3 --psm 8 '\n",
    "        r'-c tessedit_char_whitelist=ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'\n",
    "    )\n",
    "\n",
    "    raw = pytesseract.image_to_string(binary, config=config)\n",
    "    text = re.sub(r'[^A-Z0-9]', '', raw.strip().upper())\n",
    "\n",
    "    # Retry with inverted if empty\n",
    "    if not text:\n",
    "        inverted = cv2.bitwise_not(binary)\n",
    "        raw2 = pytesseract.image_to_string(inverted, config=config)\n",
    "        text = re.sub(r'[^A-Z0-9]', '', raw2.strip().upper())\n",
    "\n",
    "    return text if text else 'UNREADABLE', binary\n",
    "\n",
    "\n",
    "if plate_img is not None:\n",
    "    plate_text, binary = extract_text(plate_img)\n",
    "\n",
    "    fig, axes = plt.subplots(1, 2, figsize=(12, 4))\n",
    "    fig.suptitle('🔤 OCR Processing', fontsize=16, fontweight='bold')\n",
    "\n",
    "    axes[0].imshow(cv2.cvtColor(plate_img, cv2.COLOR_BGR2RGB))\n",
    "    axes[0].set_title('Plate Crop (Input)')\n",
    "    axes[0].axis('off')\n",
    "\n",
    "    axes[1].imshow(binary, cmap='gray')\n",
    "    axes[1].set_title('Binary (Fed to Tesseract)')\n",
    "    axes[1].axis('off')\n",
    "\n",
    "    plt.tight_layout()\n",
    "    plt.show()\n",
    "\n",
    "    print(f'\\n🔤 OCR Result: \"{plate_text}\"')\nelse:\n",
    "    print('⚠️ Skipped — no plate detected.')"
   ]
